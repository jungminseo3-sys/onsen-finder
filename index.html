function calculateScore(onsen, choices) {
let score = 0;
const reasons = [];
if (choices.region === "any" || onsen.region === choices.region) {
score += choices.region === "any" ? 1 : 4;
if (choices.region !== "any") reasons.push("선택한 여행 지역과 잘 맞습니다");
}
if (onsen.purposes.includes(choices.purpose)) {
score += 4;
reasons.push(`${labels.purpose[choices.purpose]} 목적에 적합합니다`);
}
if (choices.season === "any" || onsen.seasons.includes(choices.season)) {
score += choices.season === "any" ? 1 : 3;
if (choices.season !== "any") reasons.push(`${labels.season[choices.season]} 여행에 추천됩니다`);
}
if (onsen.moods.includes(choices.mood)) {
score += 3;
reasons.push(`${labels.mood[choices.mood]}을 즐길 수 있습니다`);
}
if (choices.mineral==="any" || (onsen.mineralTags && onsen.mineralTags.includes(choices.mineral))) {
score += choices.mineral==="any" ? 1 : 4;
if(choices.mineral!=="any") reasons.push("선호하는 온천 성분과 일치합니다");
}
return { score, reasons };
}
function recommendOnsen(choices) {
return onsens
.map(onsen => {
const result = calculateScore(onsen, choices);
return { ...onsen, ...result };
})
.sort((a, b) => b.score - a.score || a.name.localeCompare(b.name, "ko"));
}
function renderResult(ranked, choices) {
const best = ranked[0];
const reasonText = best.reasons.length

document.getElementById("resultName").textContent = best.name;
document.getElementById("resultLocation").textContent = best.location;
document.getElementById("resultFeature").textContent = best.feature;
document.getElementById("resultSeason").textContent = best.bestSeason;
document.getElementById("resultMinerals").textContent = best.minerals;
document.getElementById("resultWaterNote").textContent = best.waterNote;

const badges = [

];
document.getElementById("resultBadges").innerHTML = badges

.join("");

.join("");
emptyState.style.display = "none";
resultCard.classList.remove("show");
void resultCard.offsetWidth;

}
